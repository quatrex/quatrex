105class _ProfilingRun: 

106 """A profiling run object. 

107 

108 Parameters 

109 ---------- 

110 eventlogs : list 

111 A list of profiling events for each rank. 

112 

113 Attributes 

114 ---------- 

115 profiling_events : list[_ProfilingEvent] 

116 A list of parsed profiling events. 

117 

118 """ 

119 

120 def __init__(self, eventlogs: list[list]): 

121 """Initializes the profiling run object.""" 

122 profiling_events: list[_ProfilingEvent] = [] 

123 for rank, events in enumerate(eventlogs): 

124 for event in events: 

125 profiling_events.append(_ProfilingEvent(event, rank)) 

126 

127 self.profiling_events = profiling_events

129 def get_stats(self) -> dict: 

130 """Returns the profiling statistics. 

131 

132 This reports some statistics for each profiled function. 

133 

134 Returns 

135 ------- 

136 dict 

137 A dictionary containing the profiling statistics. 

138 

139 """ 

140 call_stats = defaultdict(list) 

141 after_barrier_stats = defaultdict(list) 

142 ranks = defaultdict(set) 

143 depths = defaultdict(set) 

144 for event in self.profiling_events: 

145 call_stats[event.label].append(event.call_time) 

146 after_barrier_stats[event.label].append(event.after_barrier_time) 

147 ranks[event.label].add(event.rank) 

148 depths[event.label].add(event.depth) 

149 

150 stats = {} 

151 for key in call_stats: 

152 call_times = xp.array(call_stats[key]) 

153 

154 num_calls = len(call_times) 

155 num_ranks = len(ranks[key]) 

156 total_call_time = float(xp.sum(call_times)) 

157 

158 stats[key] = { 

159 "num_calls": num_calls, 

160 "num_participating_ranks": num_ranks, 

161 "num_calls_per_rank": num_calls / num_ranks, 

162 "total_call_time": total_call_time, 

163 "total_call_time_per_rank": total_call_time / num_ranks, 

164 "average_call_time": float(xp.mean(call_times)), 

165 "median_call_time": float(xp.median(call_times)), 

166 "std_call_time": float(xp.std(call_times)), 

167 "min_call_time": float(xp.min(call_times)), 

168 "max_call_time": float(xp.max(call_times)), 

169 } 

170 

171 after_barrier_times = xp.array(after_barrier_stats[key]) 

172 total_after_barrier_time = float(xp.sum(after_barrier_times)) 

173 stats[key].update( 

174 { 

175 "total_after_barrier_time": total_after_barrier_time, 

176 "total_after_barrier_time_per_rank": total_after_barrier_time 

177 / num_ranks, 

178 "average_after_barrier_time": float(xp.mean(after_barrier_times)), 

179 "median_after_barrier_time": float(xp.median(after_barrier_times)), 

180 "std_after_barrier_time": float(xp.std(after_barrier_times)), 

181 "min_after_barrier_time": float(xp.min(after_barrier_times)), 

182 "max_after_barrier_time": float(xp.max(after_barrier_times)), 

183 } 

184 ) 

185 

186 return stats

294 def profile(self, label: str, level: str, comm=None): 

295 """Profiles a function and adds profiling data to the event log. 

296 

297 Notes 

298 ----- 

299 Two environment variables control the profiling behavior: 

300 - `PROFILE_LEVEL`: The profiling level for functions. The 

301 following levels are implemented: 

302 - `"off"`: The function is not profiled. 

303 - `"default"`: The function is part of the core profiling. 

304 - `"debug"`: This function only needs to be profiled for 

305 debugging purposes. 

306 - `PROFILE_COMM_SYNC`: If set to `True`, a communicator barrier 

307 is called after the profiled function to ensure that all 

308 processes are synchronized before recording the end time. 

309 Through this, differences in between processes can be 

310 better captured. 

311 

312 Parameters 

313 ---------- 

314 label : str 

315 A label for the profiled range. This is used to identify 

316 the profiled range in the profiling data. 

317 level : str 

318 The profiling level controls whether the function is 

319 profiled or not. The following levels are implemented: 

320 - `"off"`: The function is not profiled. 

321 - `"default"`: The function is part of the core profiling. 

322 - `"debug"`: This function only needs to be profiled for 

323 debugging purposes. 

324 comm : optional 

325 An optional communicator to use for synchronization 

326 

327 Returns 

328 ------- 

329 callable 

330 The wrapped function with profiling according to the 

331 specified level. 

332 

333 """ 

334 

335 def decorator(func): 

336 if _level_to_num[level] > _level_to_num[QTX_PROFILE_LEVEL]: 

337 return func 

338 

339 @wraps(func) 

340 def wrapper(*args, **kwargs): 

341 

342 # Call the function. 

343 with self.profile_range(label, level, comm): 

344 result = func(*args, **kwargs) 

345 

346 return result 

347 

348 return wrapper 

349 

350 return decorator

352 @contextmanager 

353 def profile_range(self, label: str, level: str, comm=None): 

354 """Profiles a range of code. 

355 

356 This is a context manager that profiles a range of code. 

357 

358 Parameters 

359 ---------- 

360 label : str 

361 A label for the profiled range. This is used to identify 

362 the profiled range in the profiling data. 

363 level : str 

364 The profiling level controls whether the function is 

365 profiled or not: 

366 - `"off"`: The function is not profiled. 

367 - `"default"`: The function is part of the core profiling. 

368 - `"debug"`: This function only needs to be profiled for 

369 debugging purposes. 

370 comm : optional 

371 An optional communicator to use for synchronization. 

372 comm_world is not used to not potentially deadlock. 

373 

374 Yields 

375 ------ 

376 None 

377 The context manager does not return anything. 

378 

379 """ 

380 if level not in ("off", "default", "debug"): 

381 raise ValueError(f"Invalid profiling level {level}.") 

382 

383 if _level_to_num[level] > _level_to_num[QTX_PROFILE_LEVEL]: 

384 yield 

385 return 

386 

387 if self.print_file is None: 

388 raise ValueError("Before profiling, `set_parameters` needs to be called.") 

389 

390 self.depth += 1 

391 timestamp = time.time() 

392 

393 # NOTE: We maybe need to barrier before starting the timer 

394 exception_raised = False 

395 try: 

396 if xp.__name__ == "cupy": 

397 xp.cuda.runtime.deviceSynchronize() 

398 if NVTX_AVAILABLE: 

399 xp.cuda.nvtx.RangePush(label) 

400 start_time = time.perf_counter() 

401 

402 yield 

403 

404 except Exception: 

405 exception_raised = True 

406 raise 

407 

408 finally: 

409 if xp.__name__ == "cupy": 

410 xp.cuda.runtime.deviceSynchronize() 

411 if NVTX_AVAILABLE: 

412 xp.cuda.nvtx.RangePop() 

413 

414 call_time = time.perf_counter() - start_time 

415 

416 # Do not barrier when an exception was raised to avoid potential deadlocks. 

417 if comm is not None and QTX_PROFILE_COMM_SYNC and not exception_raised: 

418 comm.barrier() 

419 after_barrier_time = time.perf_counter() - start_time 

420 else: 

421 after_barrier_time = call_time 

422 

423 self.eventlog.append( 

424 (timestamp, self.depth, label, call_time, after_barrier_time) 

425 ) 

426 

427 if comm_world.rank == 0: 

428 offset = " " * (self.depth) 

429 self.print_file.write(f"{offset}{label} : {call_time:.4f}s") 

430 if comm is not None and QTX_PROFILE_COMM_SYNC: 

431 self.print_file.write( 

432 f"{offset}{label} all : {after_barrier_time:.4f}s" 

433 ) 

434 

435 self.depth -= 1
