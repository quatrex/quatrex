24class Device: 

25 """A quantum device for electronic transport calculations. 

26 

27 Parameters 

28 ---------- 

29 config : QuatrexConfig 

30 Configuration object containing input paths, device parameters, 

31 and computational settings. 

32 

33 Attributes 

34 ---------- 

35 config : QuatrexConfig 

36 Reference to the configuration object. 

37 hamiltonians : dict 

38 Dictionary of Hamiltonian matrices indexed by (i, j, k) lattice 

39 vectors. Keys are tuples representing the lattice vector 

40 indices, values are sparse CSR matrices. 

41 overlap_matrices : dict 

42 Dictionary of overlap matrices with the same indexing as 

43 hamiltonians. For orthogonal basis sets, defaults to identity 

44 matrices. 

45 gamma_only : bool 

46 True if only the Gamma point (0,0,0) Hamiltonian is available, 

47 indicating that k-point calculations are not possible. 

48 atom_coordinates : NDArray 

49 Array of atomic coordinates. 

50 atomic_species : NDArray 

51 Array of atom symbols for each atom. NOTE: This array is always on the 

52 host since CuPy does not support string arrays. 

53 orbital_offsets : NDArray 

54 Array of cumulative orbital counts, used to map from atoms to 

55 orbitals. orbital_offsets[i] gives the starting orbital index 

56 for atom i. 

57 potential : NDArray, optional 

58 Array of electrostatic potential for each orbital. 

59 Can be either None if no potential is provided or a 1D array 

60 where the 1D index corresponds to the orbital index or the 

61 atom index depending on the shape of the provided potential. 

62 contacts : list[Contact] 

63 List of Contact objects representing the semi-infinite leads 

64 connected to this device. 

65 

66 Methods 

67 ------- 

68 apply_potential() 

69 Apply the electrostatic potential to the Hamiltonian matrices. 

70 

71 """

73 def __init__(self, config: QuatrexConfig) -> None: 

74 """Initializes a Device object from configuration.""" 

75 

76 self.config = config 

77 

78 self._init_hamiltonian() 

79 __, self.atom_coordinates, self.atomic_species = self.load_structure(config) 

80 # TODO QTBM Device/Contact currently assumes that these quantities are on the host 

81 self.atom_coordinates = get_host(self.atom_coordinates) 

82 

83 self._init_orbitals() 

84 self.potential = self.load_potential( 

85 self.config.input_dir, 

86 self.atom_coordinates, 

87 self.atomic_species, 

88 self.config.device.num_orbitals_per_atom, 

89 ) 

90 self.apply_potential() 

91 self._add_contacts() 

92 

93 if comm.rank == 0: 

94 print( 

95 f"Device initialized with {len(self.contacts)} contacts.", 

96 flush=True, 

97 )

99 @staticmethod 

100 def load_potential( 

101 input_dir: Path, 

102 atom_coordinates: NDArray, 

103 atomic_species: NDArray, 

104 num_orbitals_per_atom: dict[str, int], 

105 ) -> NDArray: 

106 """Loads electrostatic potential data from input files. 

107 

108 Attempts to load the electrostatic potential from potential.npy in the 

109 input directory. The potential can be provided either at the atomic 

110 level or at the orbital level. 

111 

112 Parameters 

113 ---------- 

114 input_dir : Path 

115 Directory containing the `potential.npy` file. 

116 atom_coordinates : NDArray 

117 Array of atomic coordinates. 

118 atomic_species : NDArray 

119 Array of atom symbols for each atom. NOTE: This array is always on the 

120 host since CuPy does not support string arrays. 

121 num_orbitals_per_atom : dict[str, int] 

122 Dictionary mapping atomic species to the number of orbitals per 

123 atom. 

124 

125 Returns 

126 ------- 

127 NDArray 

128 The electrostatic potential array. If no potential file is found, 

129 returns an array of zeros with length equal to the total number of 

130 orbitals. 

131 

132 

133 """ 

134 orbitals_per_atom = [ 

135 num_orbitals_per_atom.get(species, 1) for species in atomic_species 

136 ] 

137 num_orbitals = np.sum(np.array(orbitals_per_atom)) 

138 

139 try: 

140 potential = distributed_load(input_dir / "potential.npy") 

141 

142 # NOTE: If atom species is only 'X', then we still repeat. 

143 if potential.shape[0] == atom_coordinates.shape[0]: 

144 # Upscale the potential to the number of orbitals 

145 potential = xp.repeat(potential, orbitals_per_atom, axis=0) 

146 elif potential.shape[0] != num_orbitals: 

147 raise ValueError( 

148 "Potential shape does not match number of atoms or orbitals." 

149 ) 

150 

151 except FileNotFoundError: 

152 potential = xp.zeros(num_orbitals) 

153 

154 return potential

156 @staticmethod 

157 def load_structure( 

158 config: QuatrexConfig, 

159 ) -> tuple[NDArray, NDArray, NDArray]: 

160 """Loads the orbital coordinates, atom coordinates, and atomic 

161 species for the device. 

162 

163 Parameters 

164 ---------- 

165 config : QuatrexConfig 

166 The Quatrex configuration. 

167 

168 Returns 

169 ------- 

170 tuple[NDArray, NDArray, NDArray] 

171 The orbital coordinates, atom coordinates, and atomic 

172 species. 

173 

174 """ 

175 

176 structure_file = config.input_dir / "structure.xyz" 

177 if not structure_file.exists(): 

178 raise FileNotFoundError(f"Structure file {structure_file} not found.") 

179 lattice_vectors, atom_coordinates, atomic_species = distributed_read_xyz( 

180 structure_file 

181 ) 

182 

183 orbitals_per_atom = [ 

184 config.device.num_orbitals_per_atom.get(s, 1) for s in atomic_species 

185 ] 

186 atom_coordinates = xp.asarray(atom_coordinates) 

187 orbital_coordinates = xp.repeat(atom_coordinates, orbitals_per_atom, axis=0) 

188 

189 if config.device.construct_from_unit_cell: 

190 

191 transport_ind = "xyz".index(config.device.transport_direction) 

192 

193 orbital_coordinates = create_coordinate_grid( 

194 orbital_coordinates, 

195 config.device.num_transport_cells 

196 * config.device.neighbor_cell_cutoff[transport_ind], 

197 transport_ind, 

198 xp.asarray(lattice_vectors), 

199 ) 

200 

201 atom_coordinates = create_coordinate_grid( 

202 atom_coordinates, 

203 config.device.num_transport_cells 

204 * config.device.neighbor_cell_cutoff[transport_ind], 

205 transport_ind, 

206 xp.asarray(lattice_vectors), 

207 ) 

208 

209 atomic_species = np.concatenate( 

210 [atomic_species] 

211 * config.device.neighbor_cell_cutoff[transport_ind] 

212 * config.device.num_transport_cells 

213 ) 

214 

215 return orbital_coordinates, atom_coordinates, atomic_species

217 def _init_hamiltonian(self) -> None: 

218 """Initializes Hamiltonian and overlap matrices from files. 

219 

220 Loads sparse matrices from .h5 files in the input directory. 

221 Files should be named "hamiltonian.h5" and 

222 "overlap.h5" where the keys are strings of [i,j,k] 

223 representing lattice vector indices. 

224 

225 For missing overlap matrices, identity matrices are assumed 

226 (orthogonal basis). The (0,0,0) Hamiltonian matrix is mandatory 

227 and its absence raises an error. 

228 

229 """ 

230 

231 self.gamma_only = False 

232 

233 if not (self.config.input_dir / "hamiltonian.h5").exists(): 

234 raise ValueError("Hamiltonian matrix not found.") 

235 

236 self.hamiltonians = load_matrices( 

237 self.config, "hamiltonian", force_complex=False 

238 ) 

239 

240 for r, h_r in self.hamiltonians.items(): 

241 if not h_r.shape[0] == h_r.shape[1]: 

242 raise ValueError( 

243 f"Hamiltonian matrix at index {r} is not square. " 

244 f"Shape: {h_r.shape}" 

245 ) 

246 

247 # assert all hamiltonians are sparse matrices 

248 if not isinstance(h_r, sparse.spmatrix): 

249 raise TypeError( 

250 f"Hamiltonian matrix at index {r} is not a sparse matrix.\n" 

251 f"Matrix type: {type(h_r)}" 

252 ) 

253 

254 self.hamiltonians[r] = sparse.csr_matrix(self.hamiltonians[r]) 

255 

256 if self.hamiltonians[r].dtype in [np.complex64, np.complex128]: 

257 self.matrices_complex = True 

258 

259 if not self.hamiltonians[r].has_canonical_format: 

260 self.hamiltonians[r].sum_duplicates() 

261 self.hamiltonians[r].sort_indices() 

262 

263 size = self.hamiltonians[(0, 0, 0)].shape[0] 

264 

265 if (self.config.input_dir / "overlap.h5").exists(): 

266 self.overlap_matrices = load_matrices(self.config, "overlap") 

267 

268 for r in self.overlap_matrices: 

269 if ( 

270 self.overlap_matrices[r].shape[0] 

271 != self.overlap_matrices[r].shape[1] 

272 ): 

273 raise ValueError( 

274 f"Overlap matrix at index {r} is not square. " 

275 f"Shape: {self.overlap_matrices[r].shape}" 

276 ) 

277 

278 if self.overlap_matrices[r].shape != (size, size): 

279 raise ValueError( 

280 f"Overlap matrix at index {r} has incompatible " 

281 f"shape with Hamiltonian. Expected {(size, size)}, " 

282 f"got {self.overlap_matrices[r].shape}." 

283 ) 

284 

285 # assert all overlap_matrices are sparse matrices 

286 if not isinstance(self.overlap_matrices[r], sparse.spmatrix): 

287 raise TypeError( 

288 f"Overlap matrix at index {r} is not a sparse matrix." 

289 ) 

290 

291 self.overlap_matrices[r] = sparse.csr_matrix(self.overlap_matrices[r]) 

292 

293 if self.overlap_matrices[r].dtype in [np.complex64, np.complex128]: 

294 self.matrices_complex = True 

295 

296 if not self.overlap_matrices[r].has_canonical_format: 

297 self.overlap_matrices[r].sum_duplicates() 

298 self.overlap_matrices[r].sort_indices() 

299 

300 else: 

301 if comm.rank == 0: 

302 warnings.warn( 

303 "No overlap matrices found. Assuming identity matrix.", 

304 ) 

305 self.overlap_matrices = { 

306 (0, 0, 0): sparse.eye(size, dtype=xp.float64, format="csr") 

307 } 

308 

309 if comm.rank == 0: 

310 print(f"Loaded {len(self.hamiltonians)} Hamiltonian matrices", flush=True) 

311 print(f"Loaded {len(self.overlap_matrices)} overlap matrices", flush=True) 

312 

313 if len(self.hamiltonians) == 1: 

314 self.gamma_only = True

316 def _init_orbitals(self) -> None: 

317 """Initializes the orbital indexing system for the device. 

318 

319 Sets up the mapping between atoms and orbitals by determining 

320 how many orbitals each atom has and creating cumulative indexing 

321 arrays. 

322 

323 The number of orbitals per atom is determined from the 

324 configuration, with a default of 1 orbital per atom if not 

325 specified. 

326 

327 """ 

328 orbitals_per_atom = np.fromiter( 

329 map( 

330 defaultdict(lambda: 1, self.config.device.num_orbitals_per_atom).get, 

331 self.atomic_species, 

332 ), 

333 dtype=np.int32, 

334 ) 

335 # Create a vector with the starting orbital for each atom 

336 self.orbital_offsets = np.hstack(([0], np.cumsum(orbitals_per_atom))) 

337 

338 # TODO: THis should probably not happen directly in the Hamiltonian, 

339 # but rather during the construction of the system matrix. 

340 def apply_potential(self) -> None: 

341 """Applies electrostatic potential to device Hamiltonian.""" 

342 

343 potential = self.potential + 1e-10 

344 

345 for r, s_r in self.overlap_matrices.items(): 

346 self.hamiltonians[r] += ( 

347 s_r.multiply(potential[:, xp.newaxis]) + s_r.multiply(potential) 

348 ) / 2

350 def _add_contacts(self): 

351 """Initializes and attaches contacts to the device. 

352 

353 Creates Contact objects for each contact defined in the device 

354 configuration. Each contact represents a semi-infinite lead 

355 connected to the finite device region, providing boundary 

356 conditions for transport calculations. 

357 

358 """ 

359 

360 contacts = [] 

361 for contact_config in self.config.device.contacts: 

362 contacts.append(Contact(device=self, contact_config=contact_config)) 

363 

364 self.contacts = contacts
